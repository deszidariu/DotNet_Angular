# ========== .NET PROJECT SETUP (Generic - Latest Version) ==========

>>> dotnet new globaljson --roll-forward latestFeature
>>> dotnet new sln
>>> dotnet new webapi -o API -controllers
>>> dotnet new classlib -o Infrastructure
>>> dotnet sln add API
>>> dotnet sln add Infrastructure
>>> dotnet sln list

# ========== SETUP API PROJECT ==========

>>> cd .\API\
>>> dotnet add reference ..\Infrastructure
>>> dotnet restore
>>> dotnet build

# ========== INSTALL DATABASE PACKAGES ==========

>>> cd ..\Infrastructure\
>>> dotnet add package Microsoft.EntityFrameworkCore.SqlServer

>>> cd ..\API\
>>> dotnet add package Microsoft.EntityFrameworkCore.Design
>>> cd ..

# ========== DATABASE SETUP (SQL SERVER WITH DOCKER) ==========

>>> New-Item docker-compose.yml -Force

# Add this content to docker-compose.yml:
# 
# services:
#   sqlserver:
#     image: mcr.microsoft.com/mssql/server:2022-latest
#     environment:
#       ACCEPT_EULA: 'Y'
#       SA_PASSWORD: 'YourPassword123!'
#     ports:
#       - "1433:1433"

>>> docker compose up -d

# Add connection string in appsettings.Development.json :
#{
#  "Logging": {
#    "LogLevel": {
#      "Default": "Information",
#      "Microsoft.AspNetCore": "Warning"
#    }
#  },
#  "ConnectionStrings": {
#    "DefaultConnection": "Server=localhost,1433;Database=payment;
#        User Id=SA;Password=YourPassword123!;TrustServerCertificate=True;"
#  }
#}

# Add in Program.cs db context:
# builder.Services.AddDbContext<PersonContext>(options =>

#     options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection"));
# });

# ========== ENTITY FRAMEWORK MIGRATIONS ==========

>>> dotnet ef
>>> dotnet ef migrations add InitialCreate -s API -p Infrastructure
>>> dotnet ef database update -s API -p Infrastructure
